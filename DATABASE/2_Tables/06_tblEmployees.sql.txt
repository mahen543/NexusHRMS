CREATE TABLE tblEmployees (
    -- Column Definitions
    EmployeeId          BIGINT        NOT NULL IDENTITY(1,1),
    EmployeeCode        NVARCHAR(64)  NOT NULL,
    FirstName           NVARCHAR(64)  NOT NULL,
    MiddleName          NVARCHAR(64)      NULL,
    LastName            NVARCHAR(64)      NULL,
    Email               NVARCHAR(128) NOT NULL,
    Cmail               NVARCHAR(128) NOT NULL,
    MobileNo            NVARCHAR(16)  NOT NULL,
    Gender              NVARCHAR(8)   NOT NULL,
    DateOfJoining       DATE          NOT NULL,
    DepartmentId        BIGINT        NOT NULL,
    DesignationId       BIGINT        NOT NULL,
    EmployeeType        NVARCHAR(32)  NOT NULL,
    ReportingManagerId  BIGINT            NULL,
    CurrentSalary       DECIMAL(18,2) NOT NULL,
    ExperienceYears     DECIMAL(5,2)  NOT NULL,
    CompanyId           BIGINT        NOT NULL,
    SiteId              BIGINT        NOT NULL,
    IsPublished         BIT           NOT NULL,
    PublishedBy         NVARCHAR(128)     NULL,
    DatePublished       DATETIME2         NULL,
    DisplayOnWeb        BIT           NOT NULL,
    SortOrder           INT           NOT NULL,
    Tag                 NVARCHAR(64)      NULL,
    Comments            NVARCHAR(512)     NULL,
    IPAddress           NVARCHAR(64)  NOT NULL,
    CreatedBy           NVARCHAR(128) NOT NULL,
    DateCreated         DATETIME2     NOT NULL,
    UpdatedBy           NVARCHAR(128)     NULL,
    LastUpdated         DATETIME2         NULL,
    IsDeleted           BIT           NOT NULL,
    DeletedBy           NVARCHAR(128)     NULL,
    DateDeleted         DATETIME2         NULL,
    Id                  NVARCHAR(64)  NOT NULL,

    -- Primary, Foreign, and Unique Key Constraints
    CONSTRAINT PK_tblEmployees_EmployeeId         PRIMARY KEY (EmployeeId),
    CONSTRAINT FK_tblEmployees_DepartmentId       FOREIGN KEY (DepartmentId) REFERENCES tblDepartments(DepartmentId),
    CONSTRAINT FK_tblEmployees_DesignationId      FOREIGN KEY (DesignationId) REFERENCES tblDesignations(DesignationId),
    CONSTRAINT FK_tblEmployees_ReportingManagerId FOREIGN KEY (ReportingManagerId) REFERENCES tblEmployees(EmployeeId),
    CONSTRAINT FK_tblEmployees_CompanyId          FOREIGN KEY (CompanyId) REFERENCES tblCompanies(CompanyId),
    CONSTRAINT FK_tblEmployees_SiteId             FOREIGN KEY (SiteId) REFERENCES tblSites(SiteId),
    CONSTRAINT UK_tblEmployees_EmployeeCode       UNIQUE (EmployeeCode),
    CONSTRAINT UK_tblEmployees_Email              UNIQUE (Email),
    CONSTRAINT UK_tblEmployees_Cmail              UNIQUE (Cmail),
    CONSTRAINT UK_tblEmployees_MobileNo           UNIQUE (MobileNo),

    -- Check Constraints
    CONSTRAINT CK_tblEmployees_CurrentSalary CHECK (CurrentSalary >= 0),

    -- Default Constraints
    CONSTRAINT DF_tblEmployees_ExperienceYears DEFAULT 0                FOR ExperienceYears,
    CONSTRAINT DF_tblEmployees_IsPublished     DEFAULT 0                FOR IsPublished,
    CONSTRAINT DF_tblEmployees_DisplayOnWeb    DEFAULT 0                FOR DisplayOnWeb,
    CONSTRAINT DF_tblEmployees_SortOrder       DEFAULT 0                FOR SortOrder,
    CONSTRAINT DF_tblEmployees_IPAddress       DEFAULT '83.110.106.181' FOR IPAddress,
    CONSTRAINT DF_tblEmployees_CreatedBy       DEFAULT 'Admin'          FOR CreatedBy,
    CONSTRAINT DF_tblEmployees_DateCreated     DEFAULT GETDATE()        FOR DateCreated,
    CONSTRAINT DF_tblEmployees_IsDeleted       DEFAULT 0                FOR IsDeleted,
    CONSTRAINT DF_tblEmployees_Id              DEFAULT NEWID()          FOR Id
);
GO
